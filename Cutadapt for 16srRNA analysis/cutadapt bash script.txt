for R1 in *_R1.fastq*; do
  # Skip any files already processed in previous runs
  if [[ "$R1" == trim_* ]]; then continue; fi

  # Extract the base sample name (e.g., CFI-10, Control-S5)
  BASE="${R1%_R1.fastq*}"

  # Identify the matching R2 file across mixed compression states
  if [[ -f "${BASE}_R2.fastq.gz" ]]; then
    R2="${BASE}_R2.fastq.gz"
  elif [[ -f "${BASE}_R2.fastq" ]]; then
    R2="${BASE}_R2.fastq"
  else
    echo "Warning: No R2 found for $R1. Skipping sample."
    continue
  fi

  echo "----------------------------------------"
  echo "Working on sample: $BASE"

  # Removing the ^ anchor allows regular 5' trimming (strips primer + preceding indices)
  # Outputting as .fastq.gz normalizes the output format
  cutadapt -g CCTACGGGNGGCWGCAG -G GGACTACNVGGGTWTCTAAT \
    --discard-untrimmed \
    -o "trim_${BASE}_R1.fastq.gz" \
    -p "trim_${BASE}_R2.fastq.gz" \
    "$R1" "$R2"
done

echo "Trimming complete!"
